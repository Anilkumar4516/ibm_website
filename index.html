SELECT c.customer_name,
       SUM(o.order_total) AS total_q3_revenue,
       DENSE_RANK() OVER (ORDER BY SUM(o.order_total) DESC) AS rank
FROM customers c
JOIN orders o ON c.id = o.customer_id
WHERE o.order_date BETWEEN '2024-07-01' AND '2024-09-30'
GROUP BY c.id, c.customer_name
ORDER BY total_q3_revenue DESC LIMIT 5;
